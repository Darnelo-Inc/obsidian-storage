```yaml
---

- name: Playbook with debug info
  hosts: all
  become: yes

  vars:
	msg1: hello
	msg2: world
	msg3: ohh
	password: 123qwe
  
  tasks:
	- name: Print Secret
	  debug: 
		var: password

	- set_fact:
		full_message: "{{ msg1 }} {{ msg2 }}"
	- debug:
		var: full_message
	
	- debug:
		var: "Stop word: {{ msg3 }}"

	- debug:
		var: ansible_distribution

	- name: Save command output
	  shell:
		cmd: uptime
	  register: res
	- debug:
		var: "{{ res.stdout }}"
...
```
